DO $EF$
BEGIN
    IF NOT EXISTS(SELECT 1 FROM pg_namespace WHERE nspname = 'custom_schema') THEN
        CREATE SCHEMA custom_schema;
    END IF;
END $EF$;
CREATE TABLE IF NOT EXISTS custom_schema."__AuditMigrationsHistory" (
    migration_id character varying(150) NOT NULL,
    product_version character varying(32) NOT NULL,
    CONSTRAINT pk___audit_migrations_history PRIMARY KEY (migration_id)
);

START TRANSACTION;
CREATE TABLE custom_schema.audit_messages (
    created_on timestamp with time zone NOT NULL,
    id bigint GENERATED BY DEFAULT AS IDENTITY,
    unique_message_id uuid NOT NULL,
    message_id text,
    message_type text,
    time_sent timestamp with time zone,
    processed_at timestamp with time zone NOT NULL,
    conversation_id character varying(450),
    is_system_message boolean NOT NULL,
    status integer NOT NULL,
    sending_endpoint_name text,
    sending_endpoint_host_id uuid,
    sending_endpoint_host text,
    receiving_endpoint_name character varying(450),
    receiving_endpoint_host_id uuid,
    receiving_endpoint_host text,
    critical_time_ticks bigint,
    processing_time_ticks bigint,
    delivery_time_ticks bigint,
    headers_json text NOT NULL,
    body_text text,
    body_state integer NOT NULL,
    body_size integer NOT NULL,
    body_content_type text,
    CONSTRAINT pk_audit_messages PRIMARY KEY (created_on, id)
) PARTITION BY RANGE (created_on);

CREATE TABLE custom_schema.failed_audit_imports (
    unique_message_id uuid NOT NULL,
    failed_at timestamp with time zone NOT NULL,
    message_id text,
    headers_json text NOT NULL,
    body bytea NOT NULL,
    exception_info text,
    CONSTRAINT pk_failed_audit_imports PRIMARY KEY (unique_message_id)
);

CREATE TABLE custom_schema.saga_snapshots (
    created_on timestamp with time zone NOT NULL,
    id bigint GENERATED BY DEFAULT AS IDENTITY,
    saga_id uuid NOT NULL,
    saga_type text,
    status integer NOT NULL,
    start_time timestamp with time zone NOT NULL,
    finish_time timestamp with time zone NOT NULL,
    processed_at timestamp with time zone NOT NULL,
    endpoint text,
    state_after_change text,
    initiating_message_json text,
    outgoing_messages_json text,
    CONSTRAINT pk_saga_snapshots PRIMARY KEY (created_on, id)
) PARTITION BY RANGE (created_on);

CREATE INDEX ix_audit_messages_conversation_id ON custom_schema.audit_messages (conversation_id);

CREATE INDEX ix_audit_messages_processed_at ON custom_schema.audit_messages (processed_at);

CREATE INDEX ix_audit_messages_receiving_endpoint_name_created_on ON custom_schema.audit_messages (receiving_endpoint_name, created_on);

CREATE INDEX ix_audit_messages_time_sent ON custom_schema.audit_messages (time_sent);

CREATE INDEX ix_audit_messages_unique_message_id ON custom_schema.audit_messages (unique_message_id);

CREATE INDEX ix_failed_audit_imports_failed_at_unique_message_id ON custom_schema.failed_audit_imports (failed_at, unique_message_id);

CREATE INDEX ix_saga_snapshots_saga_id_finish_time ON custom_schema.saga_snapshots (saga_id, finish_time);
